cmake_minimum_required(VERSION 3.16)
project(XgcWorldLidar LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
set(owner "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(world_lidar STATIC ${owner}/src/world_lidar.cpp ${owner}/src/scene_conversion.cpp ${owner}/src/convex_body_conversion.cpp)
target_compile_features(world_lidar PUBLIC cxx_std_17)
target_include_directories(world_lidar PUBLIC $<BUILD_INTERFACE:${owner}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(world_lidar PUBLIC Eigen3::Eigen Threads::Threads)
set_target_properties(world_lidar PROPERTIES EXPORT_NAME Geometry)
add_library(XgcWorldLidar::Geometry ALIAS world_lidar)
option(XGC_WORLD_LIDAR_GPU "Build original GPU library" OFF)
set(export_targets world_lidar)
if(XGC_WORLD_LIDAR_GPU)
  find_package(PCL REQUIRED COMPONENTS common io filters features search kdtree)
  find_path(XGC2_SENSING_INCLUDE xgc2_sensing/gpu.h REQUIRED)
  add_library(world_lidar_gpu STATIC ${owner}/src/shared_cloud_gpu.cpp ${owner}/src/sensor_metadata.cpp)
  target_compile_features(world_lidar_gpu PUBLIC cxx_std_17)
  target_include_directories(world_lidar_gpu PUBLIC $<BUILD_INTERFACE:${owner}/include> $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS}
    PRIVATE ${XGC2_SENSING_INCLUDE})
  target_link_libraries(world_lidar_gpu PUBLIC Eigen3::Eigen ${PCL_LIBRARIES} ${CMAKE_DL_LIBS})
  set_target_properties(world_lidar_gpu PROPERTIES EXPORT_NAME Gpu)
  add_library(XgcWorldLidar::Gpu ALIAS world_lidar_gpu)
  list(APPEND export_targets world_lidar_gpu)
endif()
install(TARGETS ${export_targets} EXPORT XgcWorldLidarTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY ${owner}/include/xgc2_world_lidar DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT XgcWorldLidarTargets NAMESPACE XgcWorldLidar:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XgcWorldLidar)
configure_package_config_file(XgcWorldLidarConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/XgcWorldLidarConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XgcWorldLidar)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/XgcWorldLidarConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/XgcWorldLidar)
option(XGC_WORLD_LIDAR_METADATA_TEST "Test the real GPU parameter domain without requiring hardware" OFF)
if(XGC_WORLD_LIDAR_METADATA_TEST)
  enable_testing()
  add_executable(world_lidar_metadata_test sensor_metadata_test.cpp ${owner}/src/sensor_metadata.cpp)
  target_compile_features(world_lidar_metadata_test PRIVATE cxx_std_17)
  target_compile_options(world_lidar_metadata_test PRIVATE -UNDEBUG)
  target_include_directories(world_lidar_metadata_test PRIVATE ${owner}/include)
  add_test(NAME world_lidar_metadata COMMAND world_lidar_metadata_test)
endif()
